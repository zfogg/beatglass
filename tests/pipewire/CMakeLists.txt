cmake_minimum_required(VERSION 3.24)
project(beatglass_pipewire_tests LANGUAGES CXX)
include(CTest)
include(FetchContent)
FetchContent_Declare(Catch2 URL https://github.com/catchorg/Catch2/archive/refs/tags/v3.8.1.tar.gz URL_HASH SHA256=18b3f70ac80fccc340d8c6ff0f339b2ae64944782f8d2fca2bd705cf47cadb79 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(Catch2)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
add_executable(pipewire_discovery_test discovery_test.cpp)
target_compile_features(pipewire_discovery_test PRIVATE cxx_std_20)
target_link_libraries(pipewire_discovery_test PRIVATE Catch2::Catch2WithMain PkgConfig::PIPEWIRE)
include(Catch)
catch_discover_tests(pipewire_discovery_test)
