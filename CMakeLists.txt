option(BEATGLASS_ENABLE_PACKAGING "Enable CPack package metadata for release candidates" OFF)

# Packaging is opt-in: normal development builds never claim to produce an
# installer.  When enabled, CPack receives reproducible package identity and
# platform-appropriate generator defaults; signing/notarization remain
# explicit native release gates documented in docs/packaging-qualification.md.
if(BEATGLASS_ENABLE_PACKAGING)
  set(CPACK_PACKAGE_NAME "beatglass")
  set(CPACK_PACKAGE_VENDOR "Beatglass")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Beatglass desktop audio analysis")
  set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
  set(CPACK_PACKAGE_CONTACT "beatglass maintainers")
  if(WIN32)
    set(CPACK_GENERATOR "ZIP")
  elseif(APPLE)
    set(CPACK_GENERATOR "DragNDrop")
  else()
    set(CPACK_GENERATOR "TGZ")
  endif()
  include(CPack)
endif()

if(BEATGLASS_BUILD_LIBRARY_PROTOTYPE)
  find_package(SQLite3 QUIET)
  if(NOT SQLite3_FOUND AND BEATGLASS_FETCH_SQLITE3)
    include(FetchContent)
    FetchContent_Declare(
      beatglass_sqlite3
      URL https://www.sqlite.org/2024/sqlite-amalgamation-3450300.zip
      URL_HASH SHA256=EA170E73E447703E8359308CA2E4366A3AE0C4304A8665896F068C736781C651
    )
    FetchContent_Populate(beatglass_sqlite3)
    add_library(beatglass_sqlite3 STATIC "${beatglass_sqlite3_SOURCE_DIR}/sqlite3.c")
    target_include_directories(beatglass_sqlite3 PUBLIC "${beatglass_sqlite3_SOURCE_DIR}")
    add_library(SQLite3::SQLite3 ALIAS beatglass_sqlite3)
    set(SQLite3_FOUND TRUE)
  endif()
  if(NOT SQLite3_FOUND)
    message(FATAL_ERROR "SQLite3 is required for BEATGLASS_BUILD_LIBRARY_PROTOTYPE. Install SQLite3 development files, provide a vcpkg toolchain, or configure -DBEATGLASS_FETCH_SQLITE3=ON to use the pinned amalgamation.")
  endif()
  add_library(beatglass_library_prototype STATIC lib/library/analysis_coordinator.cpp lib/library/job_executor.cpp lib/library/sqlite_cache.cpp lib/library/catalog.cpp lib/library/details.cpp lib/library/analysis_runner.cpp lib/library/analysis_commit.cpp lib/library/analysis_admission.cpp lib/library/storage_owner.cpp)
  target_compile_features(beatglass_library_prototype PUBLIC cxx_std_20)
  target_include_directories(beatglass_library_prototype PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
  target_link_libraries(beatglass_library_prototype PRIVATE SQLite3::SQLite3)
  if(BEATGLASS_BUILD_LIBRARY_FINGERPRINT)
    find_package(OpenSSL REQUIRED)
    target_sources(beatglass_library_prototype PRIVATE lib/library/file_fingerprint.cpp lib/library/cache_identity.cpp)
    target_link_libraries(beatglass_library_prototype PRIVATE OpenSSL::Crypto)
  endif()
endif()

  # SDK no longer supplies AGL. Apple removed the inert 64-bit framework in
  # SDK 26; retain the OpenGL link and drop only the unavailable AGL entry.
  if(APPLE AND TARGET WrapOpenGL::WrapOpenGL)
    if(IS_ABSOLUTE "${CMAKE_OSX_SYSROOT}")
      set(_beatglass_sdk "${CMAKE_OSX_SYSROOT}")
    else()
      execute_process(COMMAND xcrun --sdk macosx --show-sdk-path OUTPUT_VARIABLE _beatglass_sdk OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _beatglass_sdk_result)
    endif()
    if(_beatglass_sdk AND (NOT DEFINED _beatglass_sdk_result OR _beatglass_sdk_result EQUAL 0))
      file(GLOB _beatglass_agl "${_beatglass_sdk}/System/Library/Frameworks/AGL.framework/AGL*" "${_beatglass_sdk}/System/Library/Frameworks/AGL.framework/Versions/A/AGL*")
      if(NOT _beatglass_agl)
        get_target_property(_beatglass_gl_links WrapOpenGL::WrapOpenGL INTERFACE_LINK_LIBRARIES)
        if(_beatglass_gl_links)
          list(FILTER _beatglass_gl_links EXCLUDE REGEX "(^-framework AGL$|(^|/)AGL\\.framework(/.*)?$)")
          set_property(TARGET WrapOpenGL::WrapOpenGL PROPERTY INTERFACE_LINK_LIBRARIES "${_beatglass_gl_links}")
        endif()
      endif()
    endif()
  endif()
