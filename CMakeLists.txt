cmake_minimum_required(VERSION 3.24)
project(beatglass VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(beatglass_core STATIC lib/core/snapshot_publisher.cpp)
target_compile_features(beatglass_core PUBLIC cxx_std_20)
target_include_directories(beatglass_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
if(APPLE)
  enable_language(OBJCXX)
  add_library(beatglass_coreaudio STATIC lib/capture/coreaudio/device_discovery.mm)
  target_compile_features(beatglass_coreaudio PUBLIC cxx_std_20)
  target_link_libraries(beatglass_coreaudio PUBLIC beatglass_core "-framework CoreAudio" "-framework CoreFoundation")
endif()
if(WIN32)
  add_library(beatglass_windows_endpoints STATIC src/desktop/platform/windows_endpoints.cpp src/desktop/platform/windows_loopback_session.cpp lib/capture/wasapi/windows_loopback.cpp)
  target_include_directories(beatglass_windows_endpoints PUBLIC src/desktop/platform)
  target_link_libraries(beatglass_windows_endpoints PUBLIC beatglass_core)
  target_compile_features(beatglass_windows_endpoints PUBLIC cxx_std_20)
  target_link_libraries(beatglass_windows_endpoints PUBLIC ole32 uuid propsys)
endif()
option(BEATGLASS_BUILD_SLICE "Build decoder and aubio CLI" OFF)
option(BEATGLASS_BUILD_DESKTOP "Build Qt Quick metadata shell" OFF)
option(BEATGLASS_TEST_V1_CODECS "Require AIFF, FLAC and MP3 decoder integration tests" ON)
if(BEATGLASS_BUILD_DESKTOP AND NOT BEATGLASS_BUILD_SLICE)
  message(FATAL_ERROR "Desktop requires BEATGLASS_BUILD_SLICE")
endif()
if(BEATGLASS_BUILD_SLICE)
  find_package(Aubio CONFIG QUIET)
  find_package(SndFile CONFIG QUIET)
  if(TARGET Aubio::aubio)
    set(AUBIO_LIBRARY Aubio::aubio)
  else()
    find_path(AUBIO_INCLUDE_DIR aubio/aubio.h REQUIRED)
    find_library(AUBIO_LIBRARY NAMES aubio libaubio REQUIRED)
  endif()
  if(TARGET SndFile::sndfile)
    set(SNDFILE_LIBRARY SndFile::sndfile)
  else()
    find_path(SNDFILE_INCLUDE_DIR sndfile.h REQUIRED)
    find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile-1 REQUIRED)
  endif()
  add_library(beatglass_file lib/capture/file/file_reader.cpp lib/core/file_analysis.cpp lib/core/tempo_tracker.cpp lib/core/channel_reduction.cpp)
  target_link_libraries(beatglass_file PUBLIC beatglass_core PRIVATE ${AUBIO_LIBRARY} ${SNDFILE_LIBRARY})
  if(WIN32)
    target_sources(beatglass_windows_endpoints PRIVATE src/desktop/platform/windows_live_tempo.cpp src/desktop/platform/windows_default_tempo.cpp)
    target_link_libraries(beatglass_windows_endpoints PUBLIC beatglass_file)
  endif()
  if(AUBIO_INCLUDE_DIR)
    target_include_directories(beatglass_file PRIVATE ${AUBIO_INCLUDE_DIR})
  endif()
  if(SNDFILE_INCLUDE_DIR)
    target_include_directories(beatglass_file PRIVATE ${SNDFILE_INCLUDE_DIR})
  endif()
  add_executable(beatglass-cli src/cli/main.cpp)
  target_link_libraries(beatglass-cli PRIVATE beatglass_file)
  if(WIN32 AND TARGET SndFile::sndfile)
    add_custom_command(TARGET beatglass-cli POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:beatglass-cli> $<TARGET_FILE_DIR:beatglass-cli> COMMAND_EXPAND_LISTS)
  endif()
endif()
if(BEATGLASS_BUILD_DESKTOP)
  set(BEATGLASS_QT_VERSION "" CACHE STRING "Exact Qt version for a pinned SDK (empty accepts any compatible Qt 6)")
  if(BEATGLASS_QT_VERSION)
    find_package(Qt6 ${BEATGLASS_QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
  else()
    find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
  endif()
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  add_library(beatglass_desktop_metadata STATIC src/desktop/metadata.cpp src/desktop/metadata.hpp src/desktop/live_presentation.cpp src/desktop/live_presentation.hpp)
  target_include_directories(beatglass_desktop_metadata PUBLIC src/desktop)
  target_link_libraries(beatglass_desktop_metadata PUBLIC beatglass_file Qt6::Core)
  if(WIN32)
    add_library(beatglass_windows_dj_bridge STATIC src/desktop/platform/windows_dj_bridge.cpp src/desktop/platform/windows_dj_bridge.hpp)
    target_include_directories(beatglass_windows_dj_bridge PUBLIC src/desktop/platform)
    target_link_libraries(beatglass_windows_dj_bridge PUBLIC beatglass_windows_endpoints Qt6::Core)
    target_compile_features(beatglass_windows_dj_bridge PUBLIC cxx_std_20)
  endif()
  qt_add_executable(beatglass-desktop src/desktop/main.cpp src/desktop/resources.qrc)
  target_link_libraries(beatglass-desktop PRIVATE beatglass_desktop_metadata Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
  if(WIN32)
    if(NOT TARGET Qt6::windeployqt)
      message(FATAL_ERROR "Qt windeployqt is required for the Windows desktop build")
    endif()
    add_custom_command(TARGET beatglass-desktop POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:beatglass-desktop> $<TARGET_FILE_DIR:beatglass-desktop>
      COMMAND $<TARGET_FILE:Qt6::windeployqt> --no-translations --qmldir ${CMAKE_CURRENT_SOURCE_DIR}/src/desktop/qml $<TARGET_FILE:beatglass-desktop>
      COMMAND_EXPAND_LISTS VERBATIM)
  endif()
endif()
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(Catch2 URL https://github.com/catchorg/Catch2/archive/refs/tags/v3.8.1.tar.gz URL_HASH SHA256=18b3f70ac80fccc340d8c6ff0f339b2ae64944782f8d2fca2bd705cf47cadb79)
  FetchContent_MakeAvailable(Catch2)
  add_executable(beatglass_tests tests/unit/presentation_gate_test.cpp tests/unit/job_lifecycle_test.cpp tests/unit/source_lifecycle_test.cpp tests/unit/audio_queue_test.cpp tests/unit/audio_clock_test.cpp tests/unit/stream_continuity_test.cpp tests/unit/tracking_machine_test.cpp tests/unit/snapshot_publisher_test.cpp)
  target_link_libraries(beatglass_tests PRIVATE beatglass_core Catch2::Catch2WithMain)
  if(APPLE)
    target_sources(beatglass_tests PRIVATE tests/coreaudio/device_discovery_test.cpp)
    target_link_libraries(beatglass_tests PRIVATE beatglass_coreaudio)
  endif()
  if(BEATGLASS_BUILD_DESKTOP)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(live_presentation_tests tests/unit/live_presentation_test.cpp)
    target_link_libraries(live_presentation_tests PRIVATE beatglass_desktop_metadata Qt6::Test)
    add_test(NAME live_presentation COMMAND live_presentation_tests)
    set_tests_properties(live_presentation PROPERTIES LABELS unit TIMEOUT 15)
    add_executable(metadata_tests tests/unit/metadata_test.cpp)
    target_link_libraries(metadata_tests PRIVATE beatglass_desktop_metadata Qt6::Test)
    add_test(NAME desktop_navigation COMMAND Qt6::qmltestrunner
      -input ${CMAKE_CURRENT_SOURCE_DIR}/tests/qml
      -import ${CMAKE_CURRENT_SOURCE_DIR}/src/desktop/qml)
    set_tests_properties(desktop_navigation PROPERTIES LABELS unit TIMEOUT 30
      ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    add_test(NAME metadata_worker COMMAND metadata_tests)
    set_tests_properties(metadata_worker PROPERTIES LABELS unit TIMEOUT 15)
    if(WIN32)
      add_executable(windows_dj_bridge_tests tests/windows/dj_bridge_test.cpp)
      target_link_libraries(windows_dj_bridge_tests PRIVATE beatglass_windows_dj_bridge Qt6::Test winmm)
      add_test(NAME windows_dj_bridge COMMAND windows_dj_bridge_tests)
      set_tests_properties(windows_dj_bridge PROPERTIES LABELS "unit;platform" TIMEOUT 30)
      add_custom_command(TARGET windows_dj_bridge_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:windows_dj_bridge_tests> $<TARGET_FILE_DIR:windows_dj_bridge_tests> COMMAND_EXPAND_LISTS)
      add_custom_command(TARGET live_presentation_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:live_presentation_tests> $<TARGET_FILE_DIR:live_presentation_tests> COMMAND_EXPAND_LISTS)
      add_custom_command(TARGET metadata_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:metadata_tests> $<TARGET_FILE_DIR:metadata_tests> COMMAND_EXPAND_LISTS)
    endif()
  endif()
  if(BEATGLASS_BUILD_SLICE)
    target_sources(beatglass_tests PRIVATE tests/unit/tempo_tracker_test.cpp tests/unit/channel_reduction_test.cpp tests/unit/file_reader_test.cpp)
    target_link_libraries(beatglass_tests PRIVATE beatglass_file)
    if(WIN32 AND TARGET SndFile::sndfile)
      add_custom_command(TARGET beatglass_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:beatglass_tests> $<TARGET_FILE_DIR:beatglass_tests> COMMAND_EXPAND_LISTS)
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME cli_generated_audio COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/cli_audio.py $<TARGET_FILE:beatglass-cli>)
    set_tests_properties(cli_generated_audio PROPERTIES LABELS integration TIMEOUT 60)
    if(BEATGLASS_TEST_V1_CODECS)
      add_test(NAME cli_v1_codecs COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/cli_formats.py $<TARGET_FILE:beatglass-cli> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/generated)
      set_tests_properties(cli_v1_codecs PROPERTIES LABELS integration TIMEOUT 60)
    endif()
  endif()
  include(Catch)
  catch_discover_tests(beatglass_tests PROPERTIES LABELS unit TIMEOUT 30)
  if(WIN32)
    add_executable(windows_endpoint_tests tests/windows/endpoints_test.cpp tests/windows/loopback_test.cpp)
    if(BEATGLASS_BUILD_SLICE)
      target_sources(windows_endpoint_tests PRIVATE tests/windows/live_tempo_test.cpp tests/windows/default_tempo_test.cpp)
      if(TARGET SndFile::sndfile)
        add_custom_command(TARGET windows_endpoint_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:windows_endpoint_tests> $<TARGET_FILE_DIR:windows_endpoint_tests> COMMAND_EXPAND_LISTS)
      endif()
    endif()
    target_link_libraries(windows_endpoint_tests PRIVATE beatglass_windows_endpoints Catch2::Catch2WithMain winmm)
    catch_discover_tests(windows_endpoint_tests PROPERTIES LABELS "unit;platform" TIMEOUT 30)
  endif()
endif()
