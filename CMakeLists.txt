project(beatglass VERSION 0.1.0 LANGUAGES CXX)

    message(FATAL_ERROR "PipeWire support requires Linux")
  endif()

  target_compile_features(beatglass_pipewire PUBLIC cxx_std_20)
  target_include_directories(beatglass_pipewire PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)

  message(FATAL_ERROR "Desktop requires BEATGLASS_BUILD_SLICE")

if(BEATGLASS_BUILD_SLICE)
  find_package(Aubio CONFIG QUIET)
  find_package(SndFile CONFIG QUIET)
  if(TARGET Aubio::aubio)
    set(AUBIO_LIBRARY Aubio::aubio)
  else()
    find_path(AUBIO_INCLUDE_DIR aubio/aubio.h REQUIRED)
    find_library(AUBIO_LIBRARY NAMES aubio libaubio REQUIRED)
  endif()
  if(TARGET SndFile::sndfile)
    set(SNDFILE_LIBRARY SndFile::sndfile)
  else()
    find_path(SNDFILE_INCLUDE_DIR sndfile.h REQUIRED)
    find_library(SNDFILE_LIBRARY NAMES sndfile libsndfile-1 REQUIRED)
  endif()
  add_library(beatglass_file lib/capture/file/file_reader.cpp lib/capture/file/virtual_io.cpp lib/core/file_analysis.cpp lib/core/causal_replay.cpp lib/core/tempo_tracker.cpp lib/core/beat_tracker.cpp lib/core/channel_reduction.cpp)
  target_link_libraries(beatglass_file PUBLIC beatglass_core PRIVATE ${AUBIO_LIBRARY} ${SNDFILE_LIBRARY})
  if(AUBIO_INCLUDE_DIR)
    target_include_directories(beatglass_file PRIVATE ${AUBIO_INCLUDE_DIR})
  endif()
  if(SNDFILE_INCLUDE_DIR)
    target_include_directories(beatglass_file PRIVATE ${SNDFILE_INCLUDE_DIR})
  endif()
  add_executable(beatglass-cli src/cli/main.cpp)
  target_link_libraries(beatglass-cli PRIVATE beatglass_file)
  if(MINGW)
    target_link_options(beatglass-cli PRIVATE -municode)
  endif()
  if(WIN32 AND TARGET SndFile::sndfile)
    add_custom_command(TARGET beatglass-cli POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:beatglass-cli> $<TARGET_FILE_DIR:beatglass-cli> COMMAND_EXPAND_LISTS)
  endif()
endif()
if(BEATGLASS_BUILD_DESKTOP)
  set(BEATGLASS_QT_VERSION "" CACHE STRING "Exact Qt version for a pinned SDK (empty accepts any compatible Qt 6)")
  if(BEATGLASS_QT_VERSION)
    find_package(Qt6 ${BEATGLASS_QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
  else()
    find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
  endif()
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  add_library(beatglass_desktop_metadata STATIC src/desktop/metadata.cpp src/desktop/metadata.hpp src/desktop/live_presentation.cpp src/desktop/live_presentation.hpp)
  target_include_directories(beatglass_desktop_metadata PUBLIC src/desktop)
  target_link_libraries(beatglass_desktop_metadata PUBLIC beatglass_file Qt6::Core)
  qt_add_executable(beatglass-desktop src/desktop/main.cpp src/desktop/resources.qrc)
  target_link_libraries(beatglass-desktop PRIVATE beatglass_desktop_metadata Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
  if(WIN32)
    if(NOT TARGET Qt6::windeployqt)
      message(FATAL_ERROR "Qt windeployqt is required for the Windows desktop build")
    endif()
    add_custom_command(TARGET beatglass-desktop POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:beatglass-desktop> $<TARGET_FILE_DIR:beatglass-desktop>

      COMMAND_EXPAND_LISTS VERBATIM)
  endif()
endif()
